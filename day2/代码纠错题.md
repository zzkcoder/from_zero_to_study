def outer():
    x = 1
    def inner():
        nonlocal x
        x += 1
        return x
    return inner

f = outer()
print(f(), f())

def d1(f):
    def w():
        print("d1")
        return f()
    return w

def d2(f):
    def w():
        print("d2")
        return f()
    return w

@d1
@d2
def hello():
    print("hello")
